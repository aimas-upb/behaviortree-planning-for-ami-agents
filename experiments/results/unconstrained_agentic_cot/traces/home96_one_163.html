        self.action_url = action_url

class SetFanSpeed(py_trees.behaviour.Behaviour):
    def __init__(self, name, action_url, fan_speed):
        super().__init__(name)
        self.action_url = action_url
        self.fan_speed = fan_speed

    def update(self):
        try:
            response = http_client.post(self.action_url, json={"fan_speed": self.fan_speed})
            if response.status_code == 200:
                return Status.SUCCESS
            return Status.FAILURE
        except Exception:
            return Status.FAILURE
