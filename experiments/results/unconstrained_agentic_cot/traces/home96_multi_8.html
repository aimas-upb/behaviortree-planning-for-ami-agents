class SetAromatherapyIntensity(py_trees.behaviour.Behaviour):
    def __init__(self, name, intensity_url, intensity_value):
        super().__init__(name)
        self.intensity_url = intensity_url
        self.intensity_value = intensity_value

    def update(self):
        try:
            response = http_client.post(self.intensity_url, json={"intensity": self.intensity_value})
            if response.status_code == 200:
                return Status.SUCCESS
            return Status.FAILURE
        except Exception:
            return Status.FAILURE
