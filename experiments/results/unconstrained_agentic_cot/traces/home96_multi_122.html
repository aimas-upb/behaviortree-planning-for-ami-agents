class AdjustACTemperature(py_trees.behaviour.Behaviour):
    def __init__(self, name, temperature_url, action_url, adjustment):
        super().__init__(name)
        self.temperature_url = temperature_url
        self.action_url = action_url
        self.adjustment = adjustment

    def update(self):
        try:
            response = http_client.get(self.temperature_url)
            if response.status_code == 200:
                current_temp = response.json()
                new_temp = current_temp + self.adjustment
                response = http_client.post(self.action_url, json={"temperature": new_temp})
                return Status.SUCCESS if response.status_code == 200 else Status.FAILURE
            return Status.FAILURE
        except Exception:
            return Status.FAILURE
