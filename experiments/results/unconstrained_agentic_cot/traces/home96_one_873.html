class DecreaseIntensity(py_trees.behaviour.Behaviour):
    def __init__(self, name, current_intensity_url, set_intensity_url, decrease_by_percent):
        super().__init__(name)
        self.current_intensity_url = current_intensity_url
        self.set_intensity_url = set_intensity_url
        self.decrease_by_percent = decrease_by_percent

    def update(self):
        try:
            # Get the current intensity
            response = http_client.get(self.current_intensity_url)
            if response.status_code != 200:
                return Status.FAILURE

            current_intensity = response.json()
            # Calculate the new intensity
            new_intensity = max(0, current_intensity - int(current_intensity * self.decrease_by_percent / 100))

            # Set the new intensity
            response = http_client.post(self.set_intensity_url, json={"intensity": new_intensity})
            if response.status_code == 200:
                return Status.SUCCESS
            else:
                return Status.FAILURE
        except Exception:
            return Status.FAILURE
