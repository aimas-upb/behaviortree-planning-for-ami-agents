class SetTemperature(py_trees.behaviour.Behaviour):
    def __init__(self, name, temperature_url, action_url, target_temperature):
        super().__init__(name)
        self.temperature_url = temperature_url
        self.action_url = action_url
        self.target_temperature = target_temperature

    def update(self):
        try:
            # Check current temperature
            response = http_client.get(self.temperature_url)
            if response.status_code == 200:
                current_temperature = response.json()
                if current_temperature == self.target_temperature:
                    return Status.SUCCESS

            # Set the temperature
            response = http_client.post(self.action_url, json={"temperature": self.target_temperature})
            return Status.SUCCESS if response.status_code == 200 else Status.FAILURE
        except Exception:
            return Status.FAILURE
