class DecreaseVolume(py_trees.behaviour.Behaviour):
    def __init__(self, name, current_volume_url, set_volume_url, decrease_percentage):
        super().__init__(name)
        self.current_volume_url = current_volume_url
        self.set_volume_url = set_volume_url
        self.decrease_percentage = decrease_percentage

    def update(self):
        try:
            # Get the current volume
            response = http_client.get(self.current_volume_url)
            if response.status_code != 200:
                return Status.FAILURE

            current_volume = response.json()
            # Calculate the new volume
            decrease_amount = current_volume * (self.decrease_percentage / 100.0)
            new_volume = int(current_volume - decrease_amount)

            # Ensure the new volume is within valid range
            if new_volume < 0:
                new_volume = 0
            elif new_volume > 100:
                new_volume = 100

            # Set the new volume
            response = http_client.post(self.set_volume_url, json={"volume": new_volume})
            if response.status_code == 200:
                return Status.SUCCESS
            else:
                return Status.FAILURE

        except Exception:
            return Status.FAILURE
